١

تمهيد: فلسفة وجبهات الاستخدام العسكري للنظام

في خضم العمليات الميدانية والظروف القتالية الطارئة للخدمات الطبية العسكرية، يصبح الوصول السريع والآمن تماماً إلى المراجع والمناهج أمراً يترتب عليه إنقاذ حياة الجنود. يأتي تطبيق المكتبة الطبية الميدانية العسكرية (Medical Library Tactical) ليكون بوابتنا الرقمية المنيعة والمطورة لمكافحة الظروف السيئة وصيانة الأجهزة المعقدة (أجهزة تخطيط القلب، الصدمات الكهربائية الفورية، وأجهزة التنفس والإنعاش التكتيكي).

الركائز الثلاث الأساسية لبنية النظام:
  • البوابة المقررة للدراسة الأكاديمية: تسهيل قراءة الكتب، الهوامش، والامتحانات، وتقسيم وترتيب المنهج الدراسي إلى فصول منظمة (🩺 الأساسيات الطبية، 🔬 المقرر الشامل، 🧠 التخصصات العسكرية...).
  • التوطين الإكلينيكي وحساب المؤشرات: حاسب طبي متقدم لتقييم مقاييس الوعي GCS، وتصفية الكلى eGFR، وتحليل دقيق وتوطين لغازات الدم ABG مع دمج السجلات محلياً.
  • محاذاة المعدات وتقارير الصيانة: مسح ملصقات الطاقم والأشرطة الذكية عبر الكاميرا والـ QR والمطابقة الفورية للكتيب التقني.

@androidx.camera.core.ExperimentalGetImage
// محلل الصورة المصمم للتشغيل الميداني دون التسبب في تساقط الإطارات (Frames Drop)
class QrCodeAnalyzer(private val onQrScanned: (String) -> Unit) : ImageAnalysis.Analyzer {
    override fun analyze(imageProxy: ImageProxy) {
        val mediaImage = imageProxy.image
        if (mediaImage != null) {
            val image = InputImage.fromMediaImage(mediaImage, imageProxy.imageInfo.rotationDegrees)
            val scanner = BarcodeScanning.getClient()
            
            scanner.process(image)
                .addOnSuccessListener { barcodes ->
                    for (barcode in barcodes) {
                        val raw = barcode.rawValue
                        if (!raw.isNullOrEmpty()) {
                            onQrScanned(raw) // إعادة الرمز فور فك التشفير
                            break
                        }
                    }
                }
                .addOnCompleteListener { imageProxy.close() }
        } else { imageProxy.close() }
    }
}

يتم فرز هذه البيانات ومطابقتها مباشرة بالاعتماد على جدول المطابقة المرجعي لتوجيه الطبيب ومطابقة وتحديد كود الجهاز المكتشف مثل: MML-EQ-ECG بجهاز قياس تخطيط القلب alpha، لتوجيه المستخدم للقسم العلمي والمعايرات المعتمدة الفورية.

fun triggerBiometricPrompt(activity: FragmentActivity) {
    val executor = ContextCompat.getMainExecutor(activity)
    val prompt = BiometricPrompt(activity, executor, object : BiometricPrompt.AuthenticationCallback() {
        override fun onAuthenticationSucceeded(result: BiometricPrompt.AuthenticationResult) {
            onAuthSuccess?.invoke() // فك تجميد الشاشات
        }
        override fun onAuthenticationError(errCode: Int, errString: CharSequence) {
            onAuthError?.invoke(errString.toString())
        }
    })

    val infoBuilder = BiometricPrompt.PromptInfo.Builder()
        .setTitle("التحقق الأمني الميداني 🛡️")
        .setSubtitle("يرجى وضع البصمة أو استخدام وجهك أو رمز الـ PIN التكتيكي")

    // إعدادات المصادقة التوافقية (Fallback Configuration)
    val matchesCredential = BiometricManager.from(activity)
        .canAuthenticate(BiometricManager.Authenticators.DEVICE_CREDENTIAL) == BiometricManager.BIOMETRIC_SUCCESS

    if (matchesCredential) {
        infoBuilder.setAllowedAuthenticators(BiometricManager.Authenticators.BIOMETRIC_STRONG or BiometricManager.Authenticators.DEVICE_CREDENTIAL)
    } else {
        infoBuilder.setAllowedAuthenticators(BiometricManager.Authenticators.BIOMETRIC_STRONG)
        infoBuilder.setNegativeButtonText("إغلاق")
    }
    prompt.authenticate(infoBuilder.build())
}

fun generateFallbackPdf(bookTitle: String, file: File) {
    val doc = android.graphics.pdf.PdfDocument()
    
    // تدوين الصفحة الأولى (غلاف تكتيكي مزود بشعار الصليب الأحمر الإكلينيكي ومعلومات الاكتتاب الفنية)
    val pInfo = PdfDocument.PageInfo.Builder(595, 842, 1).create()
    val page = doc.startPage(pInfo)
    val canvas = page.canvas

    // صباغة الخلفية البيضاء ووضع الإطار التقني
    val bgPaint = android.graphics.Paint().apply { color = android.graphics.Color.WHITE }
    canvas.drawRect(0f, 0f, 595f, 842f, bgPaint)

    // رسم شريط الفهرس والرأسية بلون أزرق عسكري تكتيكي
    val headerPaint = android.graphics.Paint().apply { color = android.graphics.Color.parseColor("#1B314B") }
    canvas.drawRect(15f, 15f, 580f, 80f, headerPaint)

    // رسم شعار الصليب الطبي التكتيكي الأحمر في الكانفاس مباشرة
    val crossPaint = android.graphics.Paint().apply { color = android.graphics.Color.parseColor("#E74C3C") }
    canvas.drawRect(277f, 370f, 317f, 410f, crossPaint) // العمود الرأسي
    canvas.drawRect(257f, 380f, 337f, 400f, crossPaint) // العمود الأفقي

    // تدوين العنوان الفني للورقة والاسم المعتد
    val textPaint = android.graphics.Paint().apply { color = android.graphics.Color.WHITE; textSize = 14f; textAlign = Paint.Align.CENTER }
    canvas.drawText("المكتبة الطبية الميدانية العسكرية", 297f, 45f, textPaint)

    doc.finishPage(page)
    doc.writeTo(file.outputStream())
    doc.close()
}

بنية التقارير والتحليلات الطبية والتشخيص الموجه

يمثل نظام التقارير التوطينية ReportsScreen.kt وحدة ذكية تستخدم مصفوفات ومعادلات حيوية لحساب ومطابقة ثلاثة اختبارات رعاية ريادية شديدة الأهمية:

مقياس غلاسكو للوعي الغيبوبي (GCS)

يقوم بجمع قياس استجابة العين وحركة الجسم والنشاط الشفهي لتصنيف الإصابات فوراً: (إصابة حادة رعاية حرجة 💔 من 3-8، متوسطة من 9-12، وطفيفة سليمة من 13-15).

معدل الترشيح الكبيبي الكلوي (eGFR)

استعمال معادلة كوكروفت وغولت (Cockcroft-Gault Eq) لحساب كفاءة تصفية الكلى بالاعتماد على الوزن، والسن، الكرياتينين، ومضاعفات الجنس البشري.

تحليل غازات الدم الشرياني (ABG)

مكافئ تشخيصي يقيس التوازن الهيدروجيني pH، والضغط الجزئي لثاني أكسيد الكربون PaCO2، والبيكربونات HCO3 لمطابقة وتحديد حالات الحماض أو القلوية الاستقلابية والتنفسية.

fun diagnoseABG(ph: Double, paco2: Double, hco3: Double): String {
    val isAcidemia = ph < 7.35
    val isAlkalemia = ph > 7.45
    
    return when {
        isAcidemia -> {
            if (paco2 > 45) {
                if (hco3 > 26) "Respiratory Acidosis - Partially Compensated 🫁"
                else "Respiratory Acidosis - Uncompensated 🫁"
            } else {
                if (paco2 < 35) "Metabolic Acidosis - Partially Compensated 🤢"
                else "Metabolic Acidosis - Uncompensated 🤢"
            }
        }
        isAlkalemia -> {
            if (paco2 < 35) "Respiratory Alkalosis - Uncompensated 🫁"
            else "Metabolic Alkalosis - Uncompensated 🧪"
        }
        else -> "Normal Acid-Base Balance - Symmetrical ✅"
    }
}

ثانياً: معادلة ترشيح الكلى الرياضية المطبقة (Cockcroft-Gault Equation):

eGFR = [ (140 - Age) × Weight (kg) ] / [ 72 × Creatinine (mg/dL) ]
(مضروباً بالقيمة المرجعية 0.85 في حال تصنيف المريض كأنثى لملاءمة الفروق العضلية العضوية)

val shareIntent = Intent(Intent.ACTION_SEND).apply {
    type = "text/plain"
    putExtra(Intent.EXTRA_STREAM, uri)
    putExtra(Intent.EXTRA_SUBJECT, "تصدير البيانات الطبية والعلمية الميدانية 🧬")
    addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION) // ترخيص الوصول لنشاط الاستيراد الخارجي
}
context.startActivity(Intent.createChooser(shareIntent, "تصدير التقرير الموحد عبر:").addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
